  function extOf(url) {
    const m = url.split('?')[0].match(/\.(webp|jpe?g|png|gif|svg)$/i);
    return m ? m[1].toLowerCase() : 'png';

  function moodOf(url) {

  function registerImage(url, who) {
    if (urlToPath.has(url)) return urlToPath.get(url);
    const base = `images/${safe(who)}/${safe(moodOf(url))}`;
    const ext = extOf(url);
    let path = `${base}.${ext}`;
    let n = 2;
    while (pathToUrl.has(path) && pathToUrl.get(path) !== url) path = `${base}-${n++}.${ext}`;
    urlToPath.set(url, path);
    urlMeta.set(url, { who, mood: moodOf(url) });
    pathToUrl.set(path, url);
    return path;
  }

  function fetchBuffer(url) {
    return new Promise((resolve) => {
      GM_xmlhttpRequest({
        method: 'GET',
        url,
        responseType: 'blob',
        timeout: 20000,
        onload: (r) => {
          if (r.status < 200 || r.status >= 300 || !r.response) return resolve(null);
          const fr = new FileReader();
          fr.onload = () => resolve(new Uint8Array(fr.result));
          fr.onerror = () => resolve(null);
          fr.readAsArrayBuffer(r.response);
        },
        onerror: () => resolve(null),
        ontimeout: () => resolve(null),
      });
    });
  }

  function cleanInner(srcEl, outDoc) {
    const clone = srcEl.cloneNode(true);
    clone.querySelectorAll('.dlg-scene, .dlg-add').forEach((n) => n.remove());

      const s = outDoc.createElement('span');
      s.className = 'beat-label';
      s.textContent = h.textContent.trim();
      h.replaceWith(s);
    });

      const sum = [...d.children].find((c) => c.tagName === 'SUMMARY');
      if (!sum) return;
      // The site turns source newlines into <br>; a <br> before <summary> keeps the summary from being the
      // first child, which some browsers handle badly. Remove stray <br>/blank text around the summary.
      while (sum.previousSibling) {
        const p = sum.previousSibling;
        if (p.nodeName === 'BR' || (p.nodeType === 3 && !p.textContent.trim())) p.remove(); else break;
      }
      const nx = sum.nextSibling;
      if (nx && nx.nodeName === 'BR') nx.remove();
    });
    if (EXPAND_DETAILS) clone.querySelectorAll('details').forEach((d) => d.setAttribute('open', ''));
    return clone.innerHTML;
  }

    line.className = `line kind-${kind}` + (mood ? ` mood-${mood}` : '');

      img.alt = who;
      img.src = registerImage(abs, who);

    wrap.className = 'line-body';

    whoEl.className = 'who';

    const said = outDoc.createElement('div');

    line.appendChild(wrap);
    return line;
  }

            d.innerHTML = cleanInner(holder, outDoc);
          }
          body.appendChild(d);
        }
      }
    });
    article.appendChild(body);
    return article;
  }

  // ---------- reconstruct the site's editable source (<say who=".." mood="..">) ----------
  // The history modal only shows rendered output. These rules reverse the site's rendering:
  // <br> -> newline, dlg-line -> <say>, <em> -> *..*, dice links -> plain text.

  function srcInline(node) {
    let out = '';
    node.childNodes.forEach((c) => {
      if (c.nodeType === 3) { out += c.textContent; return; }
      if (c.nodeType !== 1) return;
      const tag = c.tagName.toLowerCase();
      if (tag === 'br') out += '\n';
      else if (c.classList.contains('dlg-scene') || c.classList.contains('dlg-add')) { /* render artifact: drop */ }
      else if (tag === 'a' && c.classList.contains('dice-link')) out += c.textContent;
      else if (tag === 'span' && /display:\s*none/i.test(c.getAttribute('style') || '')) out += `<span style="display:none">${c.textContent}</span>`;
      else if (tag === 'em' || tag === 'i') out += '*' + srcInline(c) + '*';
      else if (tag === 'strong' || tag === 'b') out += '**' + srcInline(c) + '**';
      else if (tag === 'details' || tag === 'summary') out += `<${tag}>` + srcInline(c) + `</${tag}>`;
      else out += srcInline(c);
    });
    return out;
  }

  function reconstructSource(entryHtml) {
    const src = new DOMParser().parseFromString(`<body>${entryHtml}</body>`, 'text/html').body.firstElementChild;
    const role = src.querySelector('.chat-history-role')?.textContent.trim() || 'unknown';
    const content = src.querySelector('.chat-history-content');
    const blocks = [];
    [...(content?.childNodes || [])].forEach((node) => {
      if (node.nodeType === 3) {
        if (node.textContent.trim()) blocks.push({ text: node.textContent.trim(), dm: false });
        return;
      }
      if (node.nodeType !== 1) return;
      if (node.classList.contains('dlg-line')) {
        const cls = [...node.classList];
        const who =
          node.querySelector('.dlg-body')?.getAttribute('data-who') ||
          node.querySelector('.dlg-portrait')?.getAttribute('aria-label') ||
          'Unknown';
        const mood = (cls.find((c) => c.startsWith('dlg-mood-')) || '').slice(9);
        const textEl = node.querySelector('.dlg-text');
        const inner = textEl ? srcInline(textEl) : '';
        blocks.push({
          text: `<say who="${who.replace(/"/g, '&quot;')}"${mood ? ` mood="${mood}"` : ''}>${inner}</say>`,
          dm: cls.includes('dlg-kind-dm'),
        });
      } else {
        let wrap = node;
        if (!node.classList.contains('dlg-narr-block')) {
          wrap = src.ownerDocument.createElement('div'); // keep the element itself (e.g. a top-level <details>)
          wrap.appendChild(node.cloneNode(true));
        }
        const t = srcInline(wrap).replace(/^\n+|\n+$/g, '');
        if (t) blocks.push({ text: t, dm: false });
      }
    });
    let text = '';
    blocks.forEach((b, i) => {
      if (i > 0) text += blocks[i - 1].dm || b.dm ? '\n' : '\n\n';
      text += b.text;
    });
    return { role, text };
  }

  const TOOLBAR_HTML =

  function openTag(el) {
    return el.cloneNode(false).outerHTML.replace(new RegExp('</' + el.tagName + '>$', 'i'), '');
  }

  // Break long inline content onto separate lines: after <br>, and around <details>/<summary>.
  function flowLines(html, pad) {
    return html
      .replace(/<br\s*\/?>/gi, '<br>\n')
      .replace(/(<details[^>]*>)/gi, '\n$1\n')
      .replace(/(<\/details>)/gi, '\n$1\n')
      .replace(/(<summary[^>]*>[\s\S]*?<\/summary>)/gi, '$1\n')
      .split('\n')
      .map((l) => l.trim())
      .filter((l) => l !== '')
      .map((l) => pad + l)
      .join('\n');
  }

  function pretty(el, depth) {
    const pad = '  '.repeat(depth);
    const tag = el.tagName.toLowerCase();
    if (el.classList.contains('said') || el.classList.contains('narration')) {
      return `${pad}${openTag(el)}\n${flowLines(el.innerHTML, pad + '  ')}\n${pad}</${tag}>`;
    }
    if (!el.children.length) return pad + el.outerHTML;
    return `${pad}${openTag(el)}\n${[...el.children].map((c) => pretty(c, depth + 1)).join('\n')}\n${pad}</${tag}>`;
  }

      '</nav>';
    const body = g.items.map((a) => pretty(a.el, 1)).join('\n');
    return `<!DOCTYPE html>
<html lang="en"><head><meta charset="utf-8"><title>${esc(title)} ${g.min}-${g.max}</title>
<meta name="viewport" content="width=device-width,initial-scale=1">

<body>

${nav}

<main>
${body}
</main>
${nav}
<script src="size.js"></script>
<script src="portraits.js"></script>
<script src="say.js"></script>
</body></html>
`;

  function saveBlob(blob, filename) {
    const b = new Blob([blob], { type: 'application/octet-stream' });
    const url = URL.createObjectURL(b);
    const cleanup = () => setTimeout(() => URL.revokeObjectURL(url), 15000);

    const anchorFallback = () => {
      const a = document.createElement('a');
      a.href = url;
      a.download = filename;
      a.rel = 'noopener';
      a.style.display = 'none';
      document.body.appendChild(a);
      a.click();
      a.remove();
      cleanup();
    };

    // .zip is on Tampermonkey's default download whitelist, so GM_download keeps the filename.
    if (typeof GM_download === 'function') {
      try {
        GM_download({
          url,
          name: filename,
          saveAs: true,
          onload: cleanup,
          onerror: (err) => {
            console.warn('GM_download failed, using fallback:', err);
            anchorFallback();
          },
        });
        return;
      } catch (e) {
        console.warn('GM_download threw, using fallback:', e);
      }
    }
    anchorFallback();
  }

  // ---------- minimal zip writer (stored, no compression) ----------

  const CRC_TABLE = (() => {
    const t = new Uint32Array(256);
    for (let n = 0; n < 256; n++) {
      let c = n;
      for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;
      t[n] = c >>> 0;
    }
    return t;
  })();

  function crc32(u8) {
    let c = 0xffffffff;
    for (let i = 0; i < u8.length; i++) c = CRC_TABLE[(c ^ u8[i]) & 0xff] ^ (c >>> 8);
    return (c ^ 0xffffffff) >>> 0;
  }

  function makeZip(files) {
    const enc = new TextEncoder();
    const local = [];
    const central = [];
    let offset = 0;
    const d = new Date();
    const dosTime = (d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1);
    const dosDate = ((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();

    for (const f of files) {
      const name = enc.encode(f.name);
      const crc = crc32(f.data);
      const size = f.data.length;

      const lh = new DataView(new ArrayBuffer(30));
      lh.setUint32(0, 0x04034b50, true);
      lh.setUint16(4, 20, true);
      lh.setUint16(6, 0x0800, true);   // UTF-8 names
      lh.setUint16(8, 0, true);        // stored
      lh.setUint16(10, dosTime, true);
      lh.setUint16(12, dosDate, true);
      lh.setUint32(14, crc, true);
      lh.setUint32(18, size, true);
      lh.setUint32(22, size, true);
      lh.setUint16(26, name.length, true);
      lh.setUint16(28, 0, true);
      local.push(lh.buffer, name, f.data);

      const ch = new DataView(new ArrayBuffer(46));
      ch.setUint32(0, 0x02014b50, true);
      ch.setUint16(4, 20, true);
      ch.setUint16(6, 20, true);
      ch.setUint16(8, 0x0800, true);
      ch.setUint16(10, 0, true);
      ch.setUint16(12, dosTime, true);
      ch.setUint16(14, dosDate, true);
      ch.setUint32(16, crc, true);
      ch.setUint32(20, size, true);
      ch.setUint32(24, size, true);
      ch.setUint16(28, name.length, true);
      ch.setUint32(42, offset, true);
      central.push(ch.buffer, name);

      offset += 30 + name.length + size;
    }

    let cdSize = 0;
    central.forEach((c) => (cdSize += c.byteLength));
    const end = new DataView(new ArrayBuffer(22));
    end.setUint32(0, 0x06054b50, true);
    end.setUint16(8, files.length, true);
    end.setUint16(10, files.length, true);
    end.setUint32(12, cdSize, true);
    end.setUint32(16, offset, true);

    return new Blob([...local, ...central, end.buffer], { type: 'application/zip' });
  }

  // ---------- range walker (dependencies injected so it can be tested) ----------
  // Never throws after the first status read: on failure it returns what it collected
  // so far plus the error, so the caller can save a partial export.

  async function walkRange(from, to, total, d) {
    const warnings = [];
    let st = d.getStatus();
    if (!st) throw new Error("Couldn't read the page status (\"Messages 1-20 of N\").");
    const pageSize = st.start === 1 || st.end < total ? st.end - st.start + 1 : 20;
    const fwdClicks = Math.floor((from - 1) / pageSize);
    const backClicks = Math.floor((total - to) / pageSize);
    const dir = backClicks < fwdClicks ? 'back' : 'fwd';

    const byN = new Map();
    let error = null;
    try {
      const startBtn = dir === 'fwd' ? 'historyFirstPage' : 'historyLastPage';
      if (!d.isDisabled(startBtn)) await d.press(startBtn);

      while (true) {
        st = d.getStatus();
        if (!st) throw new Error("Couldn't read the page status.");
        const overlaps = st.end >= from && st.start <= to;
        if (overlaps) {
          const expected = st.end - st.start + 1;
          let html = [];
          for (let t = 0; t < 3; t++) {
            html = d.collect();
            if (html.length === expected) break;
            if (t < 2) await d.sleep(500 * (t + 1));
          }
          if (html.length !== expected) {
            warnings.push(
              `Page ${st.start}-${st.end}: expected ${expected} messages but found ${html.length} (numbering on this page may be off).`
            );
          }
          html.forEach((h, i) => {
            const n = st.start + i;
            if (n >= from && n <= to && !byN.has(n)) byN.set(n, h);
          });
        }
        d.progress({ overlaps, st, count: byN.size, from, to });

        const nextBtn = dir === 'fwd' ? 'historyNextPage' : 'historyPreviousPage';
        if (dir === 'fwd' ? st.end >= to : st.start <= from) break;
        if (d.isDisabled(nextBtn)) break;
        await d.wait(overlaps);
        await d.press(nextBtn);
      }
    } catch (e) {
      error = e;
    }

    const collected = [...byN.entries()].sort((a, b) => a[0] - b[0]).map(([n, html]) => ({ n, html }));
    return { collected, error, warnings };
  }

  function missingRanges(from, to, have) {
    const out = [];
    let start = null;
    for (let n = from; n <= to + 1; n++) {
      const missing = n <= to && !have.has(n);
      if (missing && start === null) start = n;
      if (!missing && start !== null) { out.push(start === n - 1 ? `${start}` : `${start}-${n - 1}`); start = null; }
    }
    return out;
  }

  async function runExport() {

      const from = Math.max(1, Number.isFinite(fromIn) ? fromIn : 1);
      const to = Math.min(total, Number.isFinite(toIn) ? toIn : total);
      if (from > to) throw new Error(`Invalid range ${from}-${to}; the chat has ${total} messages.`);

            if ($('historyPageStatus').textContent.trim() !== prev) return;
            setStatus(`Page was slow; retrying (${attempt}/3)...`);
            await sleep(1000 * attempt);

          throw new Error('Timed out waiting for the next page (3 tries).');
        },
        wait: (overlaps) => jitter(overlaps ? PAGE_DELAY_MS : SKIP_DELAY_MS),

              : `Skipping ahead (${st.start}-${st.end})...`

      if (!collected.length) throw error || new Error('No messages found in that range.');

      const have = new Set(collected.map((c) => c.n));
      const gaps = missingRanges(from, to, have);
      const partial = !!error || gaps.length > 0;
      const minN = collected[0].n;
      const maxN = collected[collected.length - 1].n;

        const srcN = countTag(a.source);

        if (raw !== srcN) mismatches.push(`Message #${a.n}: ${raw} dropdown(s) in the original, ${srcN} in the source file.`);
      });
      if (mismatches.length) {
        notes.push(`WARNING: dropdown (<details>) counts differ for ${mismatches.length} item(s):`, ...mismatches.slice(0, 20));
        notice = notes.join('\n');
      }

      const files = [];
      const zip = {
        file: (name, data) =>
          files.push({ name, data: typeof data === 'string' ? new TextEncoder().encode(data) : data }),
      };
      const uniq = [...urlToPath.entries()];
      const queue = [...uniq];

      let done = 0;

      articles.forEach((a) => {

        groupMap.get(k).push(a);
      });
      const groups = [...groupMap.entries()]
        .sort((x, y) => x[0] - y[0])
        .map(([, items]) => {

        );
      });
      zip.file('index.html', indexPage(title, summary, groups, notice));
      zip.file('search-data.js', searchData(articles));
      const portraitIdx = {};
      urlToPath.forEach((path, url) => {
        if (failedUrls.has(url)) return;
        const meta = urlMeta.get(url);
        if (!meta) return;
        if (!portraitIdx[meta.who]) portraitIdx[meta.who] = { color: whoColor[meta.who] || '', moods: {} };
        portraitIdx[meta.who].moods[meta.mood] = path;
      });
      zip.file('portraits.js', 'window.PORTRAITS = ' + JSON.stringify(portraitIdx, null, 1) + ';\n');
      zip.file('say.js', SAY_JS);
      zip.file('size.js', SIZE_JS);
      zip.file('style.css', CSS);
      zip.file('README.txt', README);
      zip.file('images-map.json', JSON.stringify(Object.fromEntries(urlToPath), null, 2));
      if (notice) zip.file('export-notes.txt', notice + '\n');

      setStatus('Building zip...');
      const blob = makeZip(files);
      saveBlob(
        blob,
        `airealm-chat-${minN}-to-${maxN}${partial ? '-PARTIAL' : ''}-${new Date().toISOString().slice(0, 10)}.zip`
      );
      if (!partial) { setLastExport(to); updateHint(); }

      const imgNote = `${uniq.length - failedUrls.size}/${uniq.length} images`;
      setStatus(
        partial
          ? `PARTIAL export saved: messages ${minN}-${maxN} (${collected.length}), ${imgNote}. ${notes[0] || ''} See export-notes.txt in the zip.`
          : `Exported messages ${from}-${to} (${collected.length}), ${imgNote}. Unzip, then open index.html.` +
              (warnings.length ? ` ${warnings.length} warning(s), see export-notes.txt.` : '')
      );
    } catch (e) {
      console.error(e);
      setStatus('Export failed: ' + e.message);
    } finally {
      btn.disabled = false;
    }
  }
})();
